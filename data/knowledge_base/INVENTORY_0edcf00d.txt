       IDENTIFICATION DIVISION.
       PROGRAM-ID. INVENTORY.
       AUTHOR. SYSTEM.
       DATE-WRITTEN. 2024-01-01.

       ENVIRONMENT DIVISION.
       INPUT-OUTPUT SECTION.
       FILE-CONTROL.
           SELECT ITEM-FILE    ASSIGN TO 'items.dat'
               ORGANIZATION IS SEQUENTIAL.
           SELECT ORDER-FILE   ASSIGN TO 'orders.dat'
               ORGANIZATION IS SEQUENTIAL.
           SELECT STOCK-REPORT ASSIGN TO 'stock_report.txt'
               ORGANIZATION IS SEQUENTIAL.

       DATA DIVISION.
       FILE SECTION.
       FD ITEM-FILE.
       01 ITEM-RECORD.
          05 ITEM-ID         PIC X(10).
          05 ITEM-NAME       PIC X(40).
          05 ITEM-CATEGORY   PIC X(15).
          05 ITEM-QTY-HAND   PIC 9(6).
          05 ITEM-QTY-MIN    PIC 9(6).
          05 ITEM-UNIT-COST  PIC S9(7)V99 COMP-3.
          05 ITEM-UNIT-PRICE PIC S9(7)V99 COMP-3.
          05 ITEM-STATUS     PIC X(1).
             88 ITEM-ACTIVE  VALUE 'A'.
             88 ITEM-DISC    VALUE 'D'.

       FD ORDER-FILE.
       01 ORDER-RECORD.
          05 ORD-ID          PIC X(12).
          05 ORD-ITEM-ID     PIC X(10).
          05 ORD-QTY         PIC 9(5).
          05 ORD-TYPE        PIC X(1).
             88 ORD-INBOUND  VALUE 'I'.
             88 ORD-OUTBOUND VALUE 'O'.
          05 ORD-DATE        PIC 9(8).

       FD STOCK-REPORT.
       01 REPORT-LINE        PIC X(132).

       WORKING-STORAGE SECTION.
       01 WS-EOF-ITEM        PIC X VALUE 'N'.
       01 WS-EOF-ORDER       PIC X VALUE 'N'.
       01 WS-TOTAL-VALUE     PIC S9(11)V99 COMP-3 VALUE ZERO.
       01 WS-LOW-STOCK-COUNT PIC 9(4) VALUE ZERO.
       01 WS-ITEM-VALUE      PIC S9(9)V99 COMP-3.
       01 WS-REORDER-NEEDED  PIC X VALUE 'N'.
       01 WS-REPORT-LINE     PIC X(132).
       01 WS-ITEM-COUNT      PIC 9(5) VALUE ZERO.

       PROCEDURE DIVISION.
       MAIN-LOGIC.
           PERFORM OPEN-FILES.
           PERFORM READ-ITEM.
           PERFORM PROCESS-ITEMS UNTIL WS-EOF-ITEM = 'Y'.
           PERFORM PRINT-SUMMARY.
           PERFORM CLOSE-FILES.
           STOP RUN.

       OPEN-FILES.
           OPEN INPUT ITEM-FILE ORDER-FILE.
           OPEN OUTPUT STOCK-REPORT.

       READ-ITEM.
           READ ITEM-FILE
               AT END MOVE 'Y' TO WS-EOF-ITEM.

       PROCESS-ITEMS.
           IF ITEM-ACTIVE
               PERFORM EVALUATE-STOCK
               PERFORM CALCULATE-VALUE
               ADD 1 TO WS-ITEM-COUNT
           END-IF.
           PERFORM READ-ITEM.

       EVALUATE-STOCK.
           IF ITEM-QTY-HAND < ITEM-QTY-MIN
               MOVE 'Y' TO WS-REORDER-NEEDED
               ADD 1 TO WS-LOW-STOCK-COUNT
               PERFORM WRITE-REORDER-ALERT
           ELSE
               MOVE 'N' TO WS-REORDER-NEEDED
           END-IF.

       CALCULATE-VALUE.
           COMPUTE WS-ITEM-VALUE = ITEM-QTY-HAND * ITEM-UNIT-COST.
           ADD WS-ITEM-VALUE TO WS-TOTAL-VALUE.

       WRITE-REORDER-ALERT.
           STRING 'LOW STOCK: ' ITEM-ID ' - ' ITEM-NAME
                  ' QTY: ' ITEM-QTY-HAND ' MIN: ' ITEM-QTY-MIN
               DELIMITED SIZE
               INTO WS-REPORT-LINE.
           MOVE WS-REPORT-LINE TO REPORT-LINE.
           WRITE REPORT-LINE.

       APPLY-ORDER.
           READ ORDER-FILE
               AT END MOVE 'Y' TO WS-EOF-ORDER.
           IF NOT WS-EOF-ORDER = 'Y'
               EVALUATE TRUE
                   WHEN ORD-INBOUND
                       ADD ORD-QTY TO ITEM-QTY-HAND
                   WHEN ORD-OUTBOUND
                       SUBTRACT ORD-QTY FROM ITEM-QTY-HAND
               END-EVALUATE
           END-IF.

       PRINT-SUMMARY.
           STRING 'INVENTORY SUMMARY - ITEMS: ' WS-ITEM-COUNT
                  ' LOW STOCK: ' WS-LOW-STOCK-COUNT
                  ' TOTAL VALUE: ' WS-TOTAL-VALUE
               DELIMITED SIZE
               INTO WS-REPORT-LINE.
           MOVE WS-REPORT-LINE TO REPORT-LINE.
           WRITE REPORT-LINE.

       CLOSE-FILES.
           CLOSE ITEM-FILE ORDER-FILE STOCK-REPORT.
